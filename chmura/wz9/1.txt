/api/ -> api:3000
location /api/ {
    rewrite ^/api/(.*)$ /$1 break;  # jeśli /admin/healt -> /health
    proxy_pass http://api;
}

/admin/ -> admin:4000
location /admin/ {
    rewrite ^/admin/(.*)$ /$1 break;
    proxy_pass http://admin;
}


location / {
    return 200 '{"message":"To jest api lub admin"}';
    add_header Content-Type application/json;
}
